\section{Sistema de evaluación}
\begin{frame}{Instrumentos de evaluación}
    \begin{itemize}
        \item Pruebas objetivas [PO] (\textbf{60\%}) $\rightarrow$ Habrá una única prueba al finalizar el módulo.
        \item Tareas evaluables [TE] (\textbf{40\%}) $\rightarrow$ Habrá una única tarea al finalizar el módulo
    \end{itemize}

    \begin{block}{\tiny{Cálculo de la nota}}
        \vspace{-1em}
        \begin{equation*}
        N = 0.60 * PO + 0.40 * TE
        \end{equation*}
    \end{block}
\end{frame}

\begin{frame}{Resultados de aprendizaje}
    \framesubtitle{$[RA1-RA3]$}
    \begin{itemize}
        \item \textbf{RA1}: Reconoce los elementos y herramientas que intervienen en el desarrollo de un programa informático, analizando sus característicasy la fases en las que actúan hasta llegar a su puesta en funcionamiento.
        \item \textbf{RA2}: Evalúa entornos integrados de desarrollo analizando sus características para editar código fuente y generar ejecutables.
        \item \textbf{RA3}: Verifica el funcionamiento de programas diseñando y realizando pruebas.
    \end{itemize}
\end{frame}

\begin{frame}{Resultados de aprendizaje}
    \framesubtitle{$[RA4-RA6]$}
    \begin{itemize}
        \item \textbf{RA4}: Optimiza código empleando las herramientas disponibles en el entorno de desarrollo.
        \item \textbf{RA5}: Genera diagramas de clases valorando su importancia en el desarrollo de aplicaciones y empleando herramientas específicas.
        \item \textbf{RA6}: Genera diagramas de comportamiento valorando su importancia en el desarrollo de aplicaciones y empleando herramientas específicas.
    \end{itemize}
\end{frame}
